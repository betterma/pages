import fs from 'fs'; const token = process.env.GH_PUSH_TOKEN; const path = 'kline.html'; const localPath = process.argv[1]; const message = process.argv[2] || 'Update kline.html'; const text = fs.readFileSync(localPath, 'utf8'); async function gh(pathname, init = {}) { const res = await fetch(`https://api.github.com${pathname}`, { ...init, headers: { Authorization: `Bearer ${token}`, Accept: 'application/vnd.github+json', 'User-Agent': 'pages-push', 'X-GitHub-Api-Version': '2022-11-28', ...(init.headers || {}), }, }); const body = await res.json().catch(() => ({})); if (!res.ok) throw new Error(`${init.method || 'GET'} ${pathname} ${res.status} ${JSON.stringify(body).slice(0, 400)}`); return body; } const meta = await gh(`/repos/betterma/pages/contents/${path}?ref=main`); const put = await gh(`/repos/betterma/pages/contents/${path}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message, content: Buffer.from(text).toString('base64'), sha: meta.sha, branch: 'main', }), }); console.log(JSON.stringify({ ok: true, commit: put.commit && put.commit.sha, contentSha: put.content && put.content.sha, htmlUrl: put.commit && put.commit.html_url, }, null, 2));
